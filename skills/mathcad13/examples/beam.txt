# Расчёт однопролётной шарнирно опёртой балки
> Равномерно распределённая нагрузка, прямоугольное деревянное сечение. Проверка прочности и жёсткости.

## 1. Исходные данные
L.p := 6 m               // пролёт балки
q := 10 kN/m             // расчётная нагрузка
b.s := 200 mm            // ширина сечения
h.s := 400 mm            // высота сечения
R.u := 13 MPa            // расчётное сопротивление изгибу (сосна, 2 сорт)
E.m := 10 GPa            // модуль упругости древесины

## 2. Внутренние усилия
Q.max := q*L.p/2         // поперечная сила на опоре
Q.max = kN
M.max := q*L.p^2/8       // изгибающий момент в середине пролёта
M.max = kN*m

## 3. Геометрические характеристики сечения
W.x := b.s*h.s^2/6
W.x = cm^3
I.x := b.s*h.s^3/12
I.x = cm^4

## 4. Проверка прочности
σ := M.max/W.x
σ = MPa
